-- 1. Users table (пользователи)
CREATE TABLE IF NOT EXISTS users (
    id INT AUTO_INCREMENT PRIMARY KEY,
    login VARCHAR(50) NOT NULL UNIQUE,
    password VARCHAR(255) NOT NULL
);

-- 2. Profiles table (профиль)
CREATE TABLE IF NOT EXISTS profiles (
    id INT AUTO_INCREMENT PRIMARY KEY,
    user_id INT NOT NULL,
    role VARCHAR(50),
    name VARCHAR(100),
    contact_info TEXT,
    avatar VARCHAR(255),
    registered_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (user_id) REFERENCES users(id) ON DELETE CASCADE
);

-- 3. Departments table (отделы)
CREATE TABLE IF NOT EXISTS departments (
    id INT AUTO_INCREMENT PRIMARY KEY,
    name VARCHAR(100) NOT NULL,
    head_id INT,
    FOREIGN KEY (head_id) REFERENCES users(id) ON DELETE SET NULL
);

-- 4. Department members table (участники отделов)
CREATE TABLE IF NOT EXISTS department_members (
    id INT AUTO_INCREMENT PRIMARY KEY,
    department_id INT NOT NULL,
    member_id INT NOT NULL,
    FOREIGN KEY (department_id) REFERENCES departments(id) ON DELETE CASCADE,
    FOREIGN KEY (member_id) REFERENCES users(id) ON DELETE CASCADE,
    UNIQUE KEY unique_member_in_department (department_id, member_id)
);

-- 5. Tasks table (задания)
CREATE TABLE IF NOT EXISTS tasks (
    id INT AUTO_INCREMENT PRIMARY KEY,
    title VARCHAR(200) NOT NULL,
    description TEXT,
    created_by INT NOT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    deadline DATE,
    status VARCHAR(50) DEFAULT 'pending',
    FOREIGN KEY (created_by) REFERENCES users(id) ON DELETE CASCADE
);

-- 6. Task attachments table (приложения к задаче)
CREATE TABLE IF NOT EXISTS task_attachments (
    id INT AUTO_INCREMENT PRIMARY KEY,
    task_id INT NOT NULL,
    attachment_link VARCHAR(500),
    FOREIGN KEY (task_id) REFERENCES tasks(id) ON DELETE CASCADE
);

-- 7. Task members table (участники задачи)
CREATE TABLE IF NOT EXISTS task_members (
    id INT AUTO_INCREMENT PRIMARY KEY,
    task_id INT NOT NULL,
    user_id INT NOT NULL,
    FOREIGN KEY (task_id) REFERENCES tasks(id) ON DELETE CASCADE,
    FOREIGN KEY (user_id) REFERENCES users(id) ON DELETE CASCADE,
    UNIQUE KEY unique_user_in_task (task_id, user_id)
);

-- 8. Reports table (отчеты)
CREATE TABLE IF NOT EXISTS reports (
    id INT AUTO_INCREMENT PRIMARY KEY,
    task_id INT NOT NULL,
    user_id INT NOT NULL,
    date DATE NOT NULL,
    title VARCHAR(200),
    description TEXT,
    status VARCHAR(50),
    FOREIGN KEY (task_id) REFERENCES tasks(id) ON DELETE CASCADE,
    FOREIGN KEY (user_id) REFERENCES users(id) ON DELETE CASCADE
);

-- 9. Report attachments table (приложения к отчетам)
CREATE TABLE IF NOT EXISTS report_attachments (
    id INT AUTO_INCREMENT PRIMARY KEY,
    report_id INT NOT NULL,
    attachment_link VARCHAR(500),
    FOREIGN KEY (report_id) REFERENCES reports(id) ON DELETE CASCADE
);

-- Create indexes for better performance
CREATE INDEX idx_profiles_user_id ON profiles(user_id);
CREATE INDEX idx_departments_head_id ON departments(head_id);
CREATE INDEX idx_department_members_department_id ON department_members(department_id);
CREATE INDEX idx_department_members_member_id ON department_members(member_id);
CREATE INDEX idx_tasks_created_by ON tasks(created_by);
CREATE INDEX idx_tasks_status ON tasks(status);
CREATE INDEX idx_tasks_deadline ON tasks(deadline);
CREATE INDEX idx_task_attachments_task_id ON task_attachments(task_id);
CREATE INDEX idx_task_members_task_id ON task_members(task_id);
CREATE INDEX idx_task_members_user_id ON task_members(user_id);
CREATE INDEX idx_reports_task_id ON reports(task_id);
CREATE INDEX idx_reports_user_id ON reports(user_id);
CREATE INDEX idx_report_attachments_report_id ON report_attachments(report_id);

DELIMITER $$

CREATE TRIGGER createProfile
AFTER INSERT ON users
FOR EACH ROW
BEGIN
    INSERT INTO profiles (user_id, name, avatar, registered_at)
    VALUES (NEW.id, NEW.login, NOW(), NOW());
END$$

DELIMITER ;

DELIMITER $$

-- 1. Удаление вложений отчета
CREATE TRIGGER delete_report_attachments
BEFORE DELETE ON reports
FOR EACH ROW
BEGIN
    DELETE FROM report_attachments WHERE report_id = OLD.id;
END$$

-- 2. Удаление вложений задачи и связанных данных
CREATE TRIGGER delete_task_full_cleanup
BEFORE DELETE ON tasks
FOR EACH ROW
BEGIN
    -- Удаляем вложения всех отчетов задачи
    DELETE ra FROM report_attachments ra
    INNER JOIN reports r ON r.id = ra.report_id
    WHERE r.task_id = OLD.id;
    
    -- Удаляем отчеты задачи
    DELETE FROM reports WHERE task_id = OLD.id;
    
    -- Удаляем вложения задачи
    DELETE FROM task_attachments WHERE task_id = OLD.id;
    
    -- Удаляем участников задачи (опционально)
    DELETE FROM task_members WHERE task_id = OLD.id;
END$$

-- 3. Удаление при удалении пользователя (опционально)
CREATE TRIGGER delete_user_related_attachments
BEFORE DELETE ON users
FOR EACH ROW
BEGIN
    -- Удаляем вложения отчетов пользователя
    DELETE ra FROM report_attachments ra
    INNER JOIN reports r ON r.id = ra.report_id
    WHERE r.user_id = OLD.id;
    
    -- Удаляем отчеты пользователя
    DELETE FROM reports WHERE user_id = OLD.id;
    
    -- Удаляем из task_members
    DELETE FROM task_members WHERE user_id = OLD.id;
    
    -- Удаляем из department_members
    DELETE FROM department_members WHERE member_id = OLD.id;
    
    -- Обновляем departments (head_id)
    UPDATE departments SET head_id = NULL WHERE head_id = OLD.id;
END$$

DELIMITER ;

DELIMITER $$

CREATE TRIGGER add_creator_to_task_members
AFTER INSERT ON tasks
FOR EACH ROW
BEGIN
    INSERT INTO task_members (task_id, user_id)
    VALUES (NEW.id, NEW.created_by)
    ON DUPLICATE KEY UPDATE task_id = task_id;
END$$

DELIMITER ;
